Bash / Curl:

# 1. Check version
LATEST=$(curl -s "{{ base_url }}/updates/dev-001/{{ app_name_title }}/version.txt")
echo "Latest version: $LATEST"

# 2. Download binary if newer
curl -O "{{ base_url }}/updates/dev-001/{{ app_name_title }}/download"

ESP32 (Arduino C++):

HTTPClient http;
http.begin("{{ base_url }}/updates/" + getDeviceId() + "/{{ app_name_title }}/version.txt");
int httpCode = http.GET();
if (httpCode == HTTP_CODE_OK) {
    String newVersion = http.getString();
    newVersion.trim();
    if (newVersion != CURRENT_VERSION) {
